<script>

const tickers=[
"SOXL",
"SOXX",
"NVDA",
"AVGO",
"AMD",
"MU",
"TSM",
"ASML"
];

let stockData=[];

async function getQuote(ticker){

const yahoo=
`https://query1.finance.yahoo.com/v7/finance/quote?symbols=${ticker}`;

const proxy=
`https://api.allorigins.win/raw?url=${encodeURIComponent(yahoo)}`;

const res=await fetch(proxy);
const json=await res.json();

return json.quoteResponse.result[0];
}

async function loadData(){

stockData=[];

let green=0;
let red=0;

for(const ticker of tickers){

try{

const q=await getQuote(ticker);

const item={
ticker:ticker,
price:q.regularMarketPrice,
change:q.regularMarketChangePercent
};

stockData.push(item);

if(item.change>=0) green++;
else red++;

}catch(e){
console.error(ticker,e);
}
}

render(green,red);
}

function render(green,red){

const tbody=document.querySelector("#stockTable tbody");
tbody.innerHTML="";

const heat=document.getElementById("heatmap");
heat.innerHTML="";

let totalChange=0;

stockData.forEach(stock=>{

totalChange+=stock.change;

const tr=document.createElement("tr");

tr.innerHTML=`
<td>${stock.ticker}</td>
<td>$${stock.price.toFixed(2)}</td>
<td class="${stock.change>=0?'green':'red'}">
${stock.change.toFixed(2)}%
</td>
`;

tbody.appendChild(tr);

const box=document.createElement("div");

box.className="box";

box.style.background=
stock.change>=0 ? "#14532d" : "#7f1d1d";

box.innerHTML=
`${stock.ticker}<br>${stock.change.toFixed(2)}%`;

heat.appendChild(box);

});

const strength=
Math.round((green/tickers.length)*100);

document.getElementById("strength").innerText=
strength+"/100";

document.getElementById("greens").innerText=green;
document.getElementById("reds").innerText=red;

const avg=totalChange/stockData.length;

let bias="Neutral";

if(avg>1) bias="Bullish";
if(avg<-1) bias="Bearish";

document.getElementById("bias").innerText=bias;

document.getElementById("updated").innerText=
"Updated: "+new Date().toLocaleString();
}

function sortTable(col){

stockData.sort((a,b)=>{

if(col===0)
return a.ticker.localeCompare(b.ticker);

if(col===1)
return b.price-a.price;

return b.change-a.change;
});

render(
stockData.filter(x=>x.change>=0).length,
stockData.filter(x=>x.change<0).length
);
}

loadData();

setInterval(loadData,60000);
